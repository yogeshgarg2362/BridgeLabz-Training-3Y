console.log("\nQ5. Person and Student Inheritance");

function Person(name) {
    this.name = name;
}

Person.prototype.showName = function () {
    console.log(`Name: ${this.name}`);
};

function Student(name, branch) {
    Person.call(this, name);
    this.branch = branch;
}

Student.prototype = Object.create(Person.prototype);
Student.prototype.constructor = Student;

Student.prototype.showBranch = function () {
    console.log(`Branch: ${this.branch}`);
};

let student1 = new Student("Priyanshu", "CSE");

student1.showName();
student1.showBranch();

console.log(student1 instanceof Student);
console.log(student1 instanceof Person);


console.log("\nQ6. Person Faculty Professor Prototype Chain");

function PersonInfo(name) {
    this.name = name;
}

PersonInfo.prototype.showPerson = function () {
    console.log(`Person Name: ${this.name}`);
};

function Faculty(name, subject) {
    PersonInfo.call(this, name);
    this.subject = subject;
}

Faculty.prototype = Object.create(PersonInfo.prototype);
Faculty.prototype.constructor = Faculty;

Faculty.prototype.showFaculty = function () {
    console.log(`Subject: ${this.subject}`);
};

function Professor(name, subject, designation) {
    Faculty.call(this, name, subject);
    this.designation = designation;
}

Professor.prototype = Object.create(Faculty.prototype);
Professor.prototype.constructor = Professor;

Professor.prototype.showProfessor = function () {
    console.log(`Designation: ${this.designation}`);
};

let professor1 = new Professor("Dr. Sharma", "JavaScript", "Senior Professor");

professor1.showPerson();
professor1.showFaculty();
professor1.showProfessor();

console.log(professor1 instanceof Professor);
console.log(professor1 instanceof Faculty);
console.log(professor1 instanceof PersonInfo);

console.log("\nQ8. Custom myMap Method");

Array.prototype.myMap = function (callback) {
    let result = [];

    for (let i = 0; i < this.length; i++) {
        if (i in this) {
            result[i] = callback(this[i], i, this);
        }
    }

    return result;
};

console.log([1, 2, 3].myMap(num => num * 2));
console.log([1, 2, 3, 4].myMap(num => num * num));


console.log("\nQ9. ES6 Class Inheritance");

class PersonClass {
    constructor(name) {
        this.name = name;
    }

    showName() {
        console.log(`Name: ${this.name}`);
    }
}

class StudentClass extends PersonClass {
    constructor(name, branch) {
        super(name);
        this.branch = branch;
    }

    showBranch() {
        console.log(`Branch: ${this.branch}`);
    }
}

let student2 = new StudentClass("Priyanshu", "CSE");

student2.showName();
student2.showBranch();

console.log(student2 instanceof StudentClass);
console.log(student2 instanceof PersonClass);
